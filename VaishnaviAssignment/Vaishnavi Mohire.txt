# Q1. Write a Function to classify a person’s age group: Child, Teen, Adult, senior.

age = int(input("Enter your age: "))

if age>=0 and age<12:
    print(f"You are a child:{age} age")
elif age>=13 and age<19:
    print(f"You are a teen:{age} age")
elif age>=20 and age<58:
    print(f"You are a adult:{age} age")
else:
    print(f"You are senior:{age} age")

# Q8. print a pattern like this:
# *** 
# ** 
# * 
# ** 
# *

    if i == 1:
        print("***")
    elif i % 2 == 0:

# Q4. Write a function to find the sum of numbers from 1 to N. 

def sum_of_numbers(N):
    total = 0
    for i in range(1, N + 1):
        total += i
    print(total)
sum_of_numbers(10)

# Q6. Write a function to print all numbers divisible by 7 but not by 5 from 1-100.

        if num % 7 == 0 and num % 5 != 0:
