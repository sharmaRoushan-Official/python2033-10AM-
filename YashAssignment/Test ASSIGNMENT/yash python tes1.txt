#Q1. Write a Function to classify a person’s age group: Child, Teen, Adult, senior. 
def person_age():
    age1=int(input("Enter age of a person "))
    if 1<=age1<14:
        print(f"The person with age {age1} falls under : CHILD")
    elif 14<=age1<18:
        print(f"The person with age {age1} falls under : TEEN")
    elif 18<=age1<=40:
        print(f"The person with age {age1} falls under : ADULT")
    elif 40<=age1<=80:
        print(f"The person with age {age1} falls under : Senior")
    else :
        print("invalid age ")
person_age()

#Q4. Write a function to find the sum of numbers from 1 to N. 
def sum_numbers():
    num1=int(input("Enter number you want sum "))
    scor1=(0)
    for i in range(1,num1+1):
       scor1=scor1+i
        
    print(f"The sum of numbers from 1 to {num1} is : {scor1}")
